cmake_minimum_required(VERSION 3.18)
project(panel_render C CXX)
if(NOT LVGL_DIR)
    message(FATAL_ERROR "Pass -DLVGL_DIR=/path/to/lvgl at tag v9.5.0")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl)
add_executable(panel_render render.c ../../main/display_policy.c ../../main/ui_theme.c ../../main/theme_art.c ../../main/font_outfit_semibold_48.c)
target_include_directories(panel_render PRIVATE stubs ../../main)
target_compile_definitions(panel_render PRIVATE WOKWI_EMULATOR=1)
target_compile_options(panel_render PRIVATE -Wall -Wextra -Werror=implicit-function-declaration)
target_link_libraries(panel_render PRIVATE lvgl m)
